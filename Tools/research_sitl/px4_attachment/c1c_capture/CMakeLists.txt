cmake_minimum_required(VERSION 3.16)
project(research_c1c_capture_host LANGUAGES CXX)

if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
	message(FATAL_ERROR "C1C helper tests are standalone only")
endif()
option(RESEARCH_C1C_HOST_TESTS "Build synthetic capture producer/evidence tests" OFF)
if(RESEARCH_C1C_HOST_TESTS)
	if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR CMAKE_CROSSCOMPILING)
		message(FATAL_ERROR "C1C host tests require native Linux")
	endif()
	include(CTest)
	if(NOT BUILD_TESTING)
		message(FATAL_ERROR "C1C host option requires BUILD_TESTING")
	endif()
	find_package(Threads REQUIRED)
	find_package(Python3 REQUIRED COMPONENTS Interpreter)
	add_library(capture_producer STATIC producer/StateCaptureProducer.cpp)
	target_include_directories(capture_producer PUBLIC ${CMAKE_CURRENT_LIST_DIR}/producer)
	target_compile_features(capture_producer PUBLIC cxx_std_17)
	target_compile_definitions(capture_producer PUBLIC RESEARCH_SITL_STATE_CAPTURE_HOST_TEST=1)
	target_compile_options(capture_producer PRIVATE -Wall -Wextra -Wpedantic -Werror)
	target_link_libraries(capture_producer PUBLIC Threads::Threads)
	add_executable(capture_producer_test tests/StateCaptureProducerTest.cpp)
	target_link_libraries(capture_producer_test PRIVATE capture_producer)
	target_compile_options(capture_producer_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
	add_test(NAME capture_producer_test COMMAND capture_producer_test)
	add_test(NAME capture_evidence_test COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_LIST_DIR}/tests/test_check_capture.py)
	add_test(NAME capture_config_binding_test COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_LIST_DIR}/tests/test_config_binding.py)
endif()
