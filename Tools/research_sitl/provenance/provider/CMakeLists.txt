cmake_minimum_required(VERSION 3.16)
project(closed_clock_observation LANGUAGES CXX)
option(RESEARCH_BUILD_CLOSED_CLOCK "Build closed Linux host clock observations" OFF)
option(RESEARCH_CLOCK_SANITIZE "Address/undefined sanitizer checks" OFF)
if(NOT RESEARCH_BUILD_CLOSED_CLOCK)
    return()
endif()
add_executable(checked_provider CheckedProvider.cpp ProviderHarness.cpp)
target_compile_features(checked_provider PRIVATE cxx_std_17)
target_compile_definitions(checked_provider PRIVATE RESEARCH_CLOSED_CLOCK_HOST=1)
target_compile_options(checked_provider PRIVATE -Wall -Wextra -Werror -Wshadow -fstack-usage)
if(RESEARCH_CLOCK_SANITIZE)
    target_compile_options(checked_provider PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(checked_provider PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME checked_provider_contract COMMAND checked_provider --self-test)
set_tests_properties(checked_provider_contract PROPERTIES TIMEOUT 15)
