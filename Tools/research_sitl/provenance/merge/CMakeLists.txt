cmake_minimum_required(VERSION 3.16)
project(closed_receipt_merge LANGUAGES CXX)
option(RESEARCH_BUILD_CLOSED_MERGE "Build standalone closed receipt observation only" OFF)
option(RESEARCH_MERGE_SANITIZE "Address/undefined sanitizer checks" OFF)
if(NOT RESEARCH_BUILD_CLOSED_MERGE)
    return()
endif()
find_package(Threads REQUIRED)
add_executable(receipt_merge ReceiptMerge.cpp MergeHarness.cpp)
target_compile_features(receipt_merge PRIVATE cxx_std_17)
target_compile_definitions(receipt_merge PRIVATE RESEARCH_CLOSED_MERGE_HOST=1)
target_compile_options(receipt_merge PRIVATE -Wall -Wextra -Werror -Wshadow -fstack-usage)
target_link_libraries(receipt_merge PRIVATE Threads::Threads)
if(RESEARCH_MERGE_SANITIZE)
    target_compile_options(receipt_merge PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(receipt_merge PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME closed_merge_contract COMMAND receipt_merge)
set_tests_properties(closed_merge_contract PROPERTIES TIMEOUT 30)
